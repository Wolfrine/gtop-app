SUBJECT_ALIASES={'विज्ञान':'Science','science':'Science','गणित मानक':'Mathematics Standard','Mathematics (Standard)':'Mathematics Standard','गणित':'Mathematics Standard','mathematics standard':'Mathematics Standard','Mathematics':'Mathematics Standard'}
def normalize(key,value):
    if key=='subject' and isinstance(value,str):return SUBJECT_ALIASES.get(value,SUBJECT_ALIASES.get(value.lower(),value))
    if key=='board' and isinstance(value,str):return value.upper()
    if key=='paper_language' or key=='answer_language':
        return {'हिंदी':'Hindi','हिन्दी':'Hindi','अंग्रेजी':'English','अंग्रेज़ी':'English','hindi':'Hindi','english':'English'}.get(value,value)
    return value

def evaluate(case,result):
    mismatches={}
    if result.get('action')!=case['expected_action']:mismatches['action']={'expected':case['expected_action'],'actual':result.get('action')}
    for key,value in case['expected_brief'].items():
        actual=normalize(key,result.get('brief',{}).get(key))
        if actual!=value:mismatches[key]={'expected':value,'actual':actual}
    missing=set(result.get('missing',[]))
    for key in case['required_missing']:
        if key not in missing:mismatches['missing_'+key]={'expected':True,'actual':False}
    return {'passed':not mismatches,'mismatches':mismatches}

def gpu():
    try:
        p=subprocess.run(['nvidia-smi','--query-gpu=memory.used,memory.free,utilization.gpu','--format=csv,noheader,nounits'],capture_output=True,text=True,timeout=5)
        return p.stdout.strip()
    except Exception:return None
