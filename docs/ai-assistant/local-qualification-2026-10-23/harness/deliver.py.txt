"""Return verified local artifacts to the demo's owning conversation record."""
import argparse,hashlib,json,pathlib,time
from dialogue import ROOT
from jobs import Ledger,digest
import paper

def deliver(jid,source):
    owner='local-operator-sachit';ledger=Ledger();row=ledger.get(owner,jid);folder=(ROOT/'jobs'/jid).resolve();source=pathlib.Path(source).resolve()
    if not source.is_relative_to(folder):raise PermissionError('Artifact source belongs to a different job')
    dialogue=json.loads((folder/'dialogue.json').read_text(encoding='utf-8'));brief=dialogue['canonical_brief'];result=json.loads(source.read_text(encoding='utf-8'));validation=paper.validate(result,brief)
    if not validation['passed']:raise ValueError(validation)
    manifest=json.loads((folder/'output/pdf/manifest.json').read_text(encoding='utf-8'))
    if manifest['source_file_sha256']!=hashlib.sha256(source.read_bytes()).hexdigest():raise ValueError('PDFs do not belong to this paper version')
    artifacts=[]
    for name in ['question-paper.pdf','marking-scheme.pdf']:
        path=folder/'output/pdf'/name
        if not path.is_file() or path.stat().st_size<1000:raise ValueError('Missing PDF '+name)
        sha=hashlib.sha256(path.read_bytes()).hexdigest()
        if not any(a['path']==str(path) and a['sha256']==sha for a in manifest['artifacts']):raise ValueError('PDF changed after rendering')
        artifacts.append({'path':str(path),'sha256':sha,'bytes':path.stat().st_size})
    response={'job_id':jid,'owner':owner,'conversation_id':jid,'state':'review_required','content_generation_complete':True,'teacher_approved':False,'message':f"{result['subject']} ka {len(paper.all_questions(result))}-question, 80-mark paper aur worked marking scheme taiyar hain. Teacher review pending hai.",'brief':brief,'paper_source':str(source),'paper_sha256':digest(result),'validation':validation,'artifacts':artifacts,'delivered_at':time.time()}
    (folder/'conversation-result.json').write_text(json.dumps(response,ensure_ascii=False,indent=2),encoding='utf-8')
    # Final local qualification state is review_required even when generation succeeded.
    with ledger.connect() as c:c.execute('UPDATE jobs SET state=?,result=?,updated=? WHERE id=? AND owner=?',('review_required',json.dumps(response),time.time(),jid,owner))
    print(json.dumps(response,ensure_ascii=True,indent=2),flush=True)

if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('job_id');p.add_argument('paper');a=p.parse_args();deliver(a.job_id,a.paper)
